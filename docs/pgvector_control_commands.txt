velostore_ai

  ------------------------
  2) BASIC HEALTH CHECKS
  ------------------------

Check extensions:

Check number of embeddings: SELECT COUNT(*) FROM product_embeddings;

Inspect rows: SELECT product_id, updated_at FROM product_embeddings
LIMIT 5;

Check embedding dimensions: SELECT vector_dims(embedding) FROM
product_embeddings LIMIT 1;

  -----------------------------
  3) TEST SEMANTIC SIMILARITY
  -----------------------------

Self similarity test: SELECT product_id, embedding <=> embedding AS
self_distance FROM product_embeddings LIMIT 1;

Vector similarity using existing vector: SELECT product_id, 1 -
(embedding <=> ( SELECT embedding FROM product_embeddings LIMIT 1 )) AS
similarity FROM product_embeddings ORDER BY similarity DESC LIMIT 5;

  ----------------------
  4) CHECK INDEX USAGE
  ----------------------

EXPLAIN ANALYZE SELECT product_id FROM product_embeddings ORDER BY
embedding <=> ( SELECT embedding FROM product_embeddings LIMIT 1 ) LIMIT
5;

List indexes: SELECT indexname, indexdef FROM pg_indexes WHERE tablename
= ‘product_embeddings’;

Check table size: SELECT
pg_size_pretty(pg_total_relation_size(‘product_embeddings’));

  ----------------------------------
  5) REBUILD IVF INDEX (IF NEEDED)
  ----------------------------------

DROP INDEX IF EXISTS idx_product_embeddings_cosine;

CREATE INDEX idx_product_embeddings_cosine ON product_embeddings USING
ivfflat (embedding vector_cosine_ops) WITH (lists = 100);

  --------------------------------
  6) ADVANCED ANALYTICS COMMANDS
  --------------------------------

A)  Compute Average Similarity Across Dataset

SELECT AVG(1 - (a.embedding <=> b.embedding)) AS avg_similarity FROM
product_embeddings a JOIN product_embeddings b ON a.product_id <>
b.product_id;

B)  Detect Outlier Embeddings (Low Average Similarity)

SELECT a.product_id, AVG(1 - (a.embedding <=> b.embedding)) AS
avg_similarity FROM product_embeddings a JOIN product_embeddings b ON
a.product_id <> b.product_id GROUP BY a.product_id ORDER BY
avg_similarity ASC LIMIT 5;

C)  Visualize Score Distribution (Bucketed Similarity)

WITH sims AS ( SELECT 1 - (a.embedding <=> b.embedding) AS similarity
FROM product_embeddings a JOIN product_embeddings b ON a.product_id <>
b.product_id ) SELECT width_bucket(similarity, 0, 1, 10) AS bucket,
COUNT(*) AS count FROM sims GROUP BY bucket ORDER BY bucket;

D)  Scientifically Evaluate Threshold

WITH scored AS ( SELECT product_id, 1 - (embedding <=> ( SELECT
embedding FROM product_embeddings LIMIT 1 )) AS similarity FROM
product_embeddings ) SELECT COUNT() FILTER (WHERE similarity >= 0.30) AS
above_30, COUNT() FILTER (WHERE similarity >= 0.40) AS above_40, COUNT()
FILTER (WHERE similarity >= 0.50) AS above_50, COUNT() FILTER (WHERE
similarity >= 0.60) AS above_60 FROM scored;
